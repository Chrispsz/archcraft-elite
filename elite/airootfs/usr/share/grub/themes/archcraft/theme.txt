# Archcraft ELITE — tema GRUB da casa (E150 · a pele roxa no boot)
# ─────────────────────────────────────────────────────────────────────────
# O fundo cinza do tema upstream morre: o wallpaper escuro/roxo da casa
# (a MESMA onda de seda da sessão — o arquivo do skel, entregue pelos
# veículos da casa; o GRUB estica o desktop-image por default, o 16:9
# nativo cobre a tela) veste o menu de boot. A barra de seleção é o
# roxo oficial #a855f7 com a borda #cba6f7 (select_bar.png — styled box
# 9-slice: o centro uniforme estica limpo; o PNG viaja como
# select_bar.png.b64 e cada veículo decodifica com o base64 -d do
# coreutils — o git da casa fica TEXT-ONLY). Sem pixmap de janela: o
# menu é transparente sobre o wallpaper (o peso do tema é O wallpaper).
#
# GRUB_DISTRIBUTOR="Archcraft ELITE" (cravado no /etc/default/grub pela
# dose) é quem batiza os títulos gerados — o theme.txt não mexe neles.
# As fontes são o "Unknown Regular" do unicode.pf2 (o que o grub-mkconfig
# carrega com GRUB_THEME — nada de DejaVu que o boot não tem garantido).

# o fundo da casa
desktop-image: "wallpaper.png"
desktop-color: "#11111b"
terminal-font: "Unknown Regular 16"

# o título da casa (Mauve da paleta oficial — contraste medido sobre a onda)
title-text: "Archcraft ELITE"
title-color: "#cba6f7"
title-font: "Unknown Regular 16"

# o menu de boot: itens nítidos, seleção roxa
+ boot_menu {

}
